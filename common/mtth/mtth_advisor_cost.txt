minister_cost = {
    base = 0
    modifier = {
        set_temp_variable = { minister_cost_factor = 1 }
        add_to_temp_variable = { var = minister_cost_factor value = modifier@civil_leader_cost_factor }
        clamp_temp_variable = {
            var = minister_cost_factor
            min = 0.1
        }

        set_temp_variable = { minister_base_cost = 5 }
        multiply_temp_variable = { var = minister_base_cost value = minister_cost_factor }

        set_temp_variable = { minister_purchase_surcharge = generic_minster_var }
        multiply_temp_variable = { var = minister_purchase_surcharge value = 5 }
        add_to_temp_variable = { var = minister_base_cost value = minister_purchase_surcharge }
        add = minister_base_cost
    }
}

minister_cost_remove = {
    base = 0
    modifier = {
        set_temp_variable = { minister_payment = mtth:minister_cost }
        multiply_temp_variable = { var = minister_payment value = -1 }
        add = minister_payment
    }
}
